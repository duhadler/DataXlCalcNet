using System;
using System.Numerics;
using FixedPrecNet;
using ArbPrecNet;
using UserFixedPrecNet;
using UserArbPrecNet;

namespace EvaluateCS
{
    public class Program
    {
        public double[,,] test4()
        {

            int xResolution = 64;
            int yResolution = 32;
            double xmin = 0;
            double xmax = 6.28318530717959;
            double ymin = -1.5707963267949;
            double ymax = 1.5707963267949;
            double[,,] d3 = new double[3, xResolution+2, yResolution+2];
            double dx = (xmax - xmin) / xResolution;
            double dy = (ymax - ymin) / yResolution;
            double u = 0.0;
            double v = 0.0;
            for (int ix = 0; ix <= xResolution + 1; ix++)
            {
                for (int iy = 0; iy <= yResolution + 1; iy++)
                {
                    u = xmin + ix * dx;
                    v = ymin + iy * dy;
                    // Start function definition
var a = 3.0;
var b = 2.0;
var c = 1.0;
var x = a * Math.Cos(u) * Math.Cos(v);
var y = b * Math.Cos(u) * Math.Sin(v);
var z = c * Math.Sin(u);
                    // End function definition
                    d3[0, ix, iy] = x;
                    d3[1, ix, iy] = z;
                    d3[2, ix, iy] = -y;
                }
            }
            return d3;
        }
    }
}
