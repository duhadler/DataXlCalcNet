using System;
using System.Numerics;
using FixedPrecNet;
using ArbPrecNet;
using UserFixedPrecNet;
using UserArbPrecNet;

namespace EvaluateCS
{
    public class Program
    {
        public double[,] test4()
        {

            int numPoints = 1024;
            int ExtraDt = 3;
            double tmin = 0;
            double tmax = 6.28318530717959;
            double[,] d3 = new double[3, numPoints+ExtraDt+1];
            double t = tmin;
            double dt = (tmax - tmin) / (numPoints);
            for (int i = 0; i < numPoints +ExtraDt; i++)
            {
                // Start function definition
var x = (2 + Math.Cos(3 * t)) * Math.Cos(2 * t);
var z = (2 + Math.Cos(3 * t)) * Math.Sin(2 * t);
var y = Math.Sin(3 * t);
                // End function definition
                d3[0, i] = x;
                d3[1, i] = z;
                d3[2, i] = -y;
                t += dt;
            }
            return d3;
        }
    }
}
